# NutUI React - AI Assistant Guide

## Project Overview

NutUI React is a JD-style lightweight mobile React component library developed by JDCFE (JD.com Front-End Team). It supports both H5 and Mini Program development with a single codebase.

**Version**: 3.0.18
**License**: MIT
**Repository**: https://github.com/jdf2e/nutui-react
**Official Site**: https://nutui.jd.com/react/
**Package Name**: @nutui/nutui-react (H5) / @nutui/nutui-react-taro (Mini Program)

### Key Features

- 🚀 **70+ High-Quality Components** - Cover mainstream mobile scenarios
- 📖 **JD APP 11.0 Design System** - Based on JD's visual specifications
- 🍭 **Tree Shaking Support** - Import components on demand
- 💪 **TypeScript Support** - Fully typed with excellent type definitions
- 🍭 **Theme Customization** - Support custom themes (default, jmapp, jrkf, dark mode)
- 🌍 **Internationalization** - Multi-language support (zh-CN, zh-TW, en-US, id-ID, tr-TR)
- 🍭 **90%+ Test Coverage** - Ensures stability and reliability
- 📱 **Cross-Platform** - H5 + Mini Program (WeChat, JD, Alipay, etc.) via Taro

### Component Structure

```
src/packages/
├── [component-name]/
│   ├── [component].tsx          # H5 implementation
```

## Installation & Setup

### Install Package

```bash
# For H5 Projects
pnpm add @nutui/nutui-react
# or
npm install @nutui/nutui-react
# or
yarn add @nutui/nutui-react

# For Taro (Mini Program) Projects
# Package name is same, but usage differs
```

### Basic Usage

#### Method 1: Full Import (Recommended for Beginners)

```tsx
import * as React from 'react'
import * as ReactDOM from 'react-dom'
// Import all styles
import '@nutui/nutui-react/dist/style.css'
// For other themes:
// import '@nutui/nutui-react/dist/style-jmapp.css'
// import '@nutui/nutui-react/dist/style-jrkf.css'

import { Button, Input, Cell } from '@nutui/nutui-react'

function App() {
  return (
    <div>
      <Button type="primary">Click Me</Button>
      <Input placeholder="Enter text" />
    </div>
  )
}
```

#### Method 2: Manual On-Demand Import

```tsx
// Import specific component and its styles
import '@nutui/nutui-react/dist/es/packages/button/style'
import Button from '@nutui/nutui-react/dist/es/packages/button'

// Don't forget to import global theme file in entry file:
import '@nutui/nutui-react/dist/styles/themes/default.css'
```

#### Method 3: Automatic On-Demand (Vite)

Install plugin:
```bash
pnpm add vite-plugin-imp -D
```

Configure vite.config.js:
```js
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import vitePluginImp from 'vite-plugin-imp'

export default defineConfig({
  plugins: [
    react(),
    vitePluginImp({
      libList: [
        {
          libName: '@nutui/nutui-react',
          style: (name) => {
            // For CSS
            return `@nutui/nutui-react/dist/es/packages/${name.toLowerCase()}/style/css`
            // For SCSS
            // return `@nutui/nutui-react/dist/es/packages/${name.toLowerCase()}/style`
          },
          replaceOldImport: false,
          camel2DashComponentName: false,
        },
      ],
    }),
  ],
})
```

Entry file (app.js):
```tsx
import '@nutui/nutui-react/dist/styles/themes/default.css'
```

#### 2. Input - 输入框组件

**Import**
```tsx
import { Input } from '@nutui/nutui-react'
```

**Complete Example**
```tsx
import React, { useState, useRef } from 'react'
import { Input, Cell, Button } from '@nutui/nutui-react'
import { Eye, EyeSlash } from '@nutui/icons-react'

function InputDemo() {
  const [text, setText] = useState('')
  const [password, setPassword] = useState('')
  const inputRef = useRef<any>(null)

  return (
    <div>
      {/* Controlled Input */}
      <Input
        value={text}
        onChange={(val) => setText(val)}
        placeholder="Please enter text"
      />

      {/* Uncontrolled Input */}
      <Input defaultValue="Default value" placeholder="Uncontrolled" />

      {/* Input Types */}
      <Input type="text" placeholder="Text input" />
      <Input type="password" placeholder="Password input" />
      <Input type="number" placeholder="Number input" />
      <Input type="digit" placeholder="Digit input (mobile)" />
      <Input type="tel" placeholder="Telephone" />
      <Input type="email" placeholder="Email" />

      {/* With Clear Button */}
      <Input
        clearable
        value={text}
        onChange={setText}
        placeholder="Clearable input"
        onClear={() => console.log('Cleared!')}
      />

      {/* Password with Visibility Toggle */}
      <Input
        type="password"
        value={password}
        onChange={setPassword}
        placeholder="Password"
        clearable
      />

      {/* Disabled & Readonly */}
      <Input disabled value="Disabled input" />
      <Input readOnly value="Read-only input" />

      {/* Max Length with Counter */}
      <Input
        maxLength={20}
        placeholder="Max 20 characters"
        showWordLimit
      />

      {/* Alignment */}
      <Input align="left" placeholder="Left align" />
      <Input align="center" placeholder="Center align" />
      <Input align="right" placeholder="Right align" />

      {/* Format Input (Phone Number) */}
      <Input
        placeholder="Phone number"
        formatter={(value) => {
          // Format: 138-0000-0000
          return value.replace(/[^\d]/g, '')
            .replace(/(\d{3})(\d{0,4})(\d{0,4})/, (match, p1, p2, p3) => {
              return [p1, p2, p3].filter(Boolean).join('-')
            })
        }}
        formatTrigger="onChange"
      />

      {/* Format Input (Credit Card) */}
      <Input
        placeholder="Credit card"
        formatter={(value) => {
          // Format: 1234 5678 9012 3456
          return value.replace(/\s/g, '').replace(/(.{4})/g, '$1 ')
        }}
        formatTrigger="onChange"
      />

      {/* With Events */}
      <Input
        placeholder="With events"
        onChange={(val) => console.log('Changed:', val)}
        onFocus={(val) => console.log('Focused:', val)}
        onBlur={(val) => console.log('Blurred:', val)}
        onClear={() => console.log('Cleared')}
      />

      {/* Auto Focus */}
      <Input autoFocus placeholder="Auto focused" />

      {/* Using Ref */}
      <Input ref={inputRef} placeholder="With ref" />
      <Button onClick={() => inputRef.current?.focus()}>
        Focus Input
      </Button>

      {/* In Cell Layout */}
      <Cell.Group>
        <Cell title="Username">
          <Input placeholder="Enter username" align="right" />
        </Cell>
        <Cell title="Password">
          <Input type="password" placeholder="Enter password" align="right" />
        </Cell>
      </Cell.Group>
    </div>
  )
}
```

**Key Props**
| Prop | Type | Default | Description |
|------|------|---------|-------------|
| value | `string` | - | Input value (controlled) |
| defaultValue | `string` | - | Default value (uncontrolled) |
| type | `'text' \| 'password' \| 'number' \| 'digit' \| 'tel' \| 'email' \| ...` | `'text'` | Input type |
| placeholder | `string` | - | Placeholder text |
| align | `'left' \| 'center' \| 'right'` | `'left'` | Text alignment |
| disabled | `boolean` | `false` | Disabled state |
| readOnly | `boolean` | `false` | Read-only state |
| autoFocus | `boolean` | `false` | Auto focus |
| clearable | `boolean` | `false` | Show clear button |
| clearIcon | `ReactNode` | - | Custom clear icon |
| maxLength | `number \| string` | - | Max length |
| showWordLimit | `boolean` | `false` | Show character count |
| formatter | `(val: string) => string` | - | Format input value |
| formatTrigger | `'onChange' \| 'onBlur'` | - | When to trigger formatter |
| onChange | `(value: string) => void` | - | Change handler |
| onFocus | `(value: string) => void` | - | Focus handler |
| onBlur | `(value: string) => void` | - | Blur handler |
| onClear | `(value: string) => void` | - | Clear handler |

#### 4. Form / FormItem - 表单组件

**Import**
```tsx
import { Form } from '@nutui/nutui-react'
```

**Complete Example**
```tsx
import React from 'react'
import {
  Form,
  Input,
  Button,
  Toast,
  Checkbox,
  Radio,
  Switch,
  TextArea,
  type FormItemRuleWithoutValidator,
} from '@nutui/nutui-react'

function FormDemo() {
  const [form] = Form.useForm()

  // Success handler
  const onFinish = (values: any) => {
    console.log('Form values:', values)
    Toast.show({ content: 'Submit success!', icon: 'success' })
  }

  // Fail handler
  const onFinishFailed = (values: any, errors: any) => {
    console.log('Validation errors:', errors)
    Toast.show({ content: 'Please check the form', icon: 'fail' })
  }

  // Custom validator
  const phoneValidator = (
    rule: FormItemRuleWithoutValidator,
    value: string
  ) => {
    return /^1[3-9]\d{9}$/.test(value)
  }

  const passwordValidator = (
    rule: FormItemRuleWithoutValidator,
    value: string
  ) => {
    return value.length >= 6 && value.length <= 20
  }

  // Confirm password validator
  const confirmPasswordValidator = async (
    rule: FormItemRuleWithoutValidator,
    value: string
  ) => {
    const password = form.getFieldValue('password')
    if (value && value !== password) {
      return Promise.reject('Passwords do not match')
    }
    return Promise.resolve()
  }

  return (
    <Form
      form={form}
      divider
      labelPosition="left"
      starPosition="right"
      onFinish={onFinish}
      onFinishFailed={onFinishFailed}
      footer={
        <div style={{ display: 'flex', justifyContent: 'center', gap: '10px' }}>
          <Button nativeType="submit" type="primary" block>
            Submit
          </Button>
          <Button nativeType="reset" block>
            Reset
          </Button>
        </div>
      }
    >
      {/* Required Field */}
      <Form.Item
        label="Username"
        name="username"
        rules={[{ required: true, message: 'Please enter username' }]}
      >
        <Input placeholder="Enter username" clearable />
      </Form.Item>

      {/* Email Validation */}
      <Form.Item
        label="Email"
        name="email"
        rules={[
          { required: true, message: 'Please enter email' },
          { type: 'email', message: 'Invalid email format' },
        ]}
      >
        <Input type="email" placeholder="Enter email" clearable />
      </Form.Item>

      {/* Custom Validator */}
      <Form.Item
        label="Phone"
        name="phone"
        rules={[
          { required: true, message: 'Please enter phone number' },
          { validator: phoneValidator, message: 'Invalid phone number' },
        ]}
      >
        <Input type="tel" placeholder="Enter phone number" clearable />
      </Form.Item>

      {/* Password */}
      <Form.Item
        label="Password"
        name="password"
        rules={[
          { required: true, message: 'Please enter password' },
          { validator: passwordValidator, message: '6-20 characters required' },
        ]}
      >
        <Input type="password" placeholder="Enter password" clearable />
      </Form.Item>

      {/* Confirm Password */}
      <Form.Item
        label="Confirm"
        name="confirmPassword"
        rules={[
          { required: true, message: 'Please confirm password' },
          { validator: confirmPasswordValidator },
        ]}
      >
        <Input type="password" placeholder="Confirm password" clearable />
      </Form.Item>

      {/* Min/Max Length */}
      <Form.Item
        label="Bio"
        name="bio"
        rules={[
          { min: 10, message: 'At least 10 characters' },
          { max: 200, message: 'Maximum 200 characters' },
        ]}
      >
        <TextArea placeholder="Tell us about yourself" maxLength={200} />
      </Form.Item>

      {/* Pattern Validation */}
      <Form.Item
        label="Code"
        name="code"
        rules={[
          { required: true, message: 'Please enter code' },
          { pattern: /^\d{6}$/, message: 'Must be 6 digits' },
        ]}
      >
        <Input placeholder="Enter 6-digit code" maxLength={6} />
      </Form.Item>

      {/* Checkbox */}
      <Form.Item
        label="Gender"
        name="gender"
        rules={[{ required: true, message: 'Please select gender' }]}
      >
        <Radio.Group>
          <Radio value="male">Male</Radio>
          <Radio value="female">Female</Radio>
        </Radio.Group>
      </Form.Item>

      {/* Checkbox Group */}
      <Form.Item
        label="Interests"
        name="interests"
        rules={[{ required: true, message: 'Please select at least one' }]}
      >
        <Checkbox.Group>
          <Checkbox value="sports">Sports</Checkbox>
          <Checkbox value="music">Music</Checkbox>
          <Checkbox value="reading">Reading</Checkbox>
        </Checkbox.Group>
      </Form.Item>

      {/* Switch (need valuePropName) */}
      <Form.Item
        label="Subscribe"
        name="subscribe"
        valuePropName="checked"
      >
        <Switch />
      </Form.Item>

      {/* Agreement Checkbox */}
      <Form.Item
        name="agreement"
        valuePropName="checked"
        rules={[
          {
            validator: (_, value) =>
              value ? Promise.resolve() : Promise.reject('Please agree'),
          },
        ]}
      >
        <Checkbox>I agree to the terms and conditions</Checkbox>
      </Form.Item>
    </Form>
  )
}

// Programmatic Form Operations
```

## Common Patterns & Best Practices

### 1. Controlled vs Uncontrolled Components

Most form components support both patterns:

```tsx
// Controlled (recommended for complex forms)
const [value, setValue] = useState('')
<Input value={value} onChange={setValue} />

// Uncontrolled (simpler, use for simple cases)
<Input defaultValue="initial" />
```

### 2. Using Refs

```tsx
import { useRef } from 'react'
import { Input } from '@nutui/nutui-react'

const inputRef = useRef(null)

<Input ref={inputRef} />
<Button onClick={() => inputRef.current?.focus()}>Focus Input</Button>
```

### 3. Imperative API (Dialog, Toast, etc.)

```tsx
// These components have both declarative and imperative APIs

// Imperative (simpler for quick usage)
Dialog.alert({ content: 'Alert!' })
Toast.show('Message')

// Declarative (more control)
<Dialog visible={visible} onConfirm={...} />
<Toast visible={visible} content="Message" />
```

### 4. Event Handling

```tsx
// Most components support standard React events
<Button onClick={(e) => console.log('Clicked', e)} />
<Input onChange={(value) => console.log('Changed', value)} />
<Input onFocus={() => console.log('Focused')} />
<Input onBlur={() => console.log('Blurred')} />
```

### 5. Styling Components

```tsx
// Use className and style props
<Button className="my-button" style={{ marginTop: 20 }}>
  Styled Button
</Button>

// Or use CSS variables for theme
<Button style={{ '--nutui-button-border-radius': '20px' }}>
  Custom Radius
</Button>
```

## Common Issues & Solutions

### 1. Import Errors

**Problem**: Cannot find module '@nutui/nutui-react'

**Solution**: Make sure you've installed the package and imported correctly

```bash
pnpm add @nutui/nutui-react
```

```tsx
import { Button } from '@nutui/nutui-react'
import '@nutui/nutui-react/dist/style.css'
```

### 2. Styles Not Loading

**Problem**: Components render but have no styles

**Solution**: Import the CSS file in your entry point

```tsx
// In main.tsx or App.tsx
import '@nutui/nutui-react/dist/style.css'
```

### 3. Type Errors

**Problem**: TypeScript errors with component props

**Solution**: NutUI is fully typed. Check the documentation for correct prop types.

```tsx
// Incorrect
<Button type="custom">Button</Button>

// Correct
<Button type="primary">Button</Button>
```

### 4. Form Validation Not Working

**Problem**: Form validation doesn't trigger

**Solution**: Ensure Form.Item has a `name` prop and Button has `nativeType="submit"`

```tsx
<Form onFinish={onFinish}>
  <Form.Item
    name="username"  // Required for validation
    rules={[{ required: true }]}
  >
    <Input />
  </Form.Item>
  <Button nativeType="submit" type="primary">Submit</Button>
</Form>
```

### 5. Taro vs H5 Differences

**Problem**: Component behaves differently in Taro

**Solution**: Some components have Taro-specific implementations. Check doc.taro.md for differences. Import from different packages:

```tsx
// H5
import { Button } from '@nutui/nutui-react'

// Taro
import { Button } from '@nutui/nutui-react-taro'
```

### 6. px to rem Conversion

**Problem**: Need rem units for responsive design

**Solution**: Configure postcss-pxtorem but exclude NutUI:

```js
// postcss.config.js
module.exports = {
  plugins: {
    'postcss-pxtorem': {
      rootValue: 37.5,
      propList: ['*'],
      selectorBlackList: ['nut-'] // Ignore NutUI classes
    }
  }
}
```

### 7. Dialog/Toast Not Showing

**Problem**: Imperative API (Dialog.confirm, Toast.show) not working

**Solution**: Make sure you're calling them inside a function, not during render:

```tsx
// Wrong
function MyComponent() {
  Toast.show('Hello') // Don't call during render
  return <div>Content</div>
}

// Correct
function MyComponent() {
  return (
    <Button onClick={() => Toast.show('Hello')}>
      Show Toast
    </Button>
  )
}
```

### 8. Checkbox/Radio Group Value Not Updating

**Problem**: Selected values not updating

**Solution**: For Checkbox.Group, use array of values:

```tsx
// Correct
<Checkbox.Group value={['1', '2']} onChange={setValues}>
  <Checkbox value="1">Option 1</Checkbox>
  <Checkbox value="2">Option 2</Checkbox>
</Checkbox.Group>

// For Radio.Group, use single value
<Radio.Group value="1" onChange={setValue}>
  <Radio value="1">Option 1</Radio>
  <Radio value="2">Option 2</Radio>
</Radio.Group>
```

## Best Practices & Tips

### 1. Component Import Strategy

**For Small Projects (< 10 components)**
- Use full import for simplicity
```tsx
import '@nutui/nutui-react/dist/style.css'
import { Button, Input, Cell } from '@nutui/nutui-react'
```

**For Large Projects**
- Use automatic on-demand import with build plugins
- Reduces bundle size significantly

### 2. Form Handling

**Prefer Controlled Components for Complex Forms**
```tsx
const [form] = Form.useForm()

// Access form values
const values = form.getFieldsValue()

// Set form values
form.setFieldsValue({ username: 'john' })

// Validate manually
await form.submit()
```

**Use Async Validators for Server-Side Validation**
```tsx
<Form.Item
  name="username"
  rules={[{
    validator: async (rule, value) => {
      const exists = await checkUsernameExists(value)
      if (exists) {
        return Promise.reject('Username already exists')
      }
      return Promise.resolve()
    }
  }]}
>
  <Input />
</Form.Item>
```

### 6. Error Handling Patterns

**Global Error Boundary**
```tsx
class ErrorBoundary extends React.Component {
  state = { hasError: false }
  
  static getDerivedStateFromError(error) {
    return { hasError: true }
  }
  
  render() {
    if (this.state.hasError) {
      return <Empty description="Something went wrong" />
    }
    return this.props.children
  }
}
```

**Async Error Handling**
```tsx
const handleSubmit = async (values) => {
  try {
    Toast.loading('Submitting...')
    await api.submit(values)
    Toast.hide()
    Toast.success('Success!')
    navigate('/success')
  } catch (error) {
    Toast.hide()
    if (error.code === 'NETWORK_ERROR') {
      Toast.fail('Network error, please try again')
    } else {
      Toast.fail(error.message || 'Submit failed')
    }
  }
}
```

```tsx
import React, { useState, useEffect } from 'react'
import { Swiper } from '@nutui/nutui-react'

function SwiperDemo() {
  const list = [
    'https://storage.360buyimg.com/jdc-article/NutUItaro34.jpg',

  return (

      <Swiper

        loop
        indicator
        style={{ height: '150px' }}
      >
        {list.map((item, index) => (
          <Swiper.Item key={item}>
            <img
              src={item}
              alt={`slide-${index}`}
              style={{ width: '100%', height: '100%' }}
              draggable={false}
            />
          </Swiper.Item>
        ))}
      </Swiper>

      {/* Custom Slide Size */}
      <Swiper
        loop

            <img src={item} draggable={false} />

      </Swiper>

      {/* Focus Effect */}
      <Swiper

        slideSize={300}

        ))}
      </Swiper>

      {/* Vertical Swiper */}
      <Swiper
        direction="vertical"
        autoPlay
        loop

// Swiper with Change Event
function SwiperWithChange() {
  const [current, setCurrent] = useState(0)
  const list = ['Slide 1', 'Slide 2', 'Slide 3']

  return (
    <div>
      <div>Current: {current + 1} / {list.length}</div>
      <Swiper
        indicator
        onChange={(index) => {
          console.log('Changed to:', index)
          setCurrent(index)
        }}
      >
        {list.map((item) => (
          <Swiper.Item key={item}>
            <div style={{
              textAlign: 'center',
              padding: '50px',
              backgroundColor: '#f5f5f5',
            }}>
              {item}
            </div>
          </Swiper.Item>
        ))}
      </Swiper>
    </div>
  )
}

// Controlled Swiper
function ControlledSwiper() {
  const [current, setCurrent] = useState(0)
  const list = ['Slide 1', 'Slide 2', 'Slide 3']

  return (
    <div>
      <Button onClick={() => setCurrent((current - 1 + list.length) % list.length)}>
        Prev
      </Button>
      <Button onClick={() => setCurrent((current + 1) % list.length)}>
        Next
      </Button>
      
      <Swiper
        value={current}
        onChange={setCurrent}
        indicator
      >
        {list.map((item) => (
          <Swiper.Item key={item}>
            <div style={{ textAlign: 'center', padding: '50px' }}>
              {item}
            </div>
          </Swiper.Item>
        ))}
      </Swiper>
    </div>
  )
}

// Async Loading
function AsyncSwiper() {
  const [list, setList] = useState<string[]>([])

  useEffect(() => {
    // Simulate async data loading
    setTimeout(() => {
      setList([
        'https://storage.360buyimg.com/jdc-article/NutUItaro34.jpg',
        'https://storage.360buyimg.com/jdc-article/NutUItaro2.jpg',
      ])
    }, 1000)
  }, [])

  if (list.length === 0) {
    return <div>Loading...</div>
  }

  return (
    <Swiper autoPlay loop indicator>
      {list.map((item) => (
        <Swiper.Item key={item}>
          <img src={item} draggable={false} />
        </Swiper.Item>
      ))}
    </Swiper>
  )
}
```

**Key Props**
| Prop | Type | Default | Description |
|------|------|---------|-------------|
| value | `number` | `0` | Current slide index (controlled) |
| defaultValue | `number` | `0` | Default slide index |
| autoPlay | `boolean \| number` | `false` | Auto play (true or duration in ms) |
| loop | `boolean` | `false` | Enable loop |
| direction | `'horizontal' \| 'vertical'` | `'horizontal'` | Swipe direction |
| indicator | `boolean` | `false` | Show indicator |
| slideSize | `number` | - | Slide width/height |
| slidesPerView | `number` | `1` | Number of slides per view |
| spaceBetween | `number` | `0` | Space between slides |
| effect | `{ name: 'focus', scale: number }` | - | Transition effect |
| onChange | `(index: number) => void` | - | Slide change handler |
